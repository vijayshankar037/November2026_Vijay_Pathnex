# Day 06 - Introdution to Cloudformation and Kubernates Advance Concepts

## Ansible - Install Docker

- name: Install Docker
  hosts: all
  become: yes

  tasks:
    - name: Install Docker
      yum:
        name: docker
        state: present

    - name: Start Docker
      service:
        name: docker
        state: started
        enabled: yes


## Terraform - Provisioning EC2 with VPC

resource "aws_vpc" "main" {
  cidr_block = "10.0.0.0/16"

  tags = {
    Name = "Pathnex-VPC"
  }
}

# Added an AWS subnet because your EC2 instance references aws_subnet.main.id,
# But the original code did not define that subnet.
resource "aws_subnet" "main" {
  vpc_id     = aws_vpc.main.id
  cidr_block = "10.0.1.0/24"

  tags = {
    Name = "Pathnex-Subnet"
  }
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-01234abcd1234abcd"
  instance_type = "t3.medium"
  subnet_id     = aws_subnet.main.id

  tags = {
    Name = "Pathnex-EC2"
  }
}


## Kubernets - Pod with volume Mount

apiVersion: v1
kind: Pod

metadata:
  name: pathnex-pod

spec:
  containers:
    - name: web
      image: nginx
      volumeMounts:
        - mountPath: /usr/share/nginx/html
          name: html-volume

  volumes:
    - name: html-volume
      hostPath:
        path: /data
        type: Directory



# Shell Script - Check for Running Services

#!/bin/bash

services=("nginx" "docker" "httpd")

for service in "${services[@]}"; do
    if systemctl is-active --quiet "$service"; then
        echo "$service is running"
    else
        echo "$service is not running"
    fi
done
