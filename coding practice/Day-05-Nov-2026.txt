## Jenkins - With Git integration and Testing

#Groovy
pipeline {
    agent any

    stages {
        stage('Clone Repository') {
            steps {
                git 'https://github.com/vijayshankar037/November2026_Vijay_Pathnex.git'
            }
        }

        stage('Build') {
            steps {
                echo 'Building project from Git repository'
            }
        }

        stage('Test') {
            steps {
                echo 'Running unit tests'
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}

#Ansible - Create User and set Permissions

---
- name: Create a new user and set permissions
  hosts: all
  become: yes

  tasks:
    - name: Create user
      user:
        name: pathnex
        state: present
        shell: /bin/bash

    - name: Set permissions for the user
      file:
        path: /home/pathnex
        owner: pathnex
        group: pathnex
        mode: '0755'
